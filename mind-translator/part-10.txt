function setPill(text){$('nodePill').textContent=text}
function scrollTop(){const box=$('dialogueScroll');box.scrollTop=0}
function nextNode(){
 const s=SCENARIOS[state.scenarioIndex];
 if(state.nodeIndex < s.nodes.length-1){state.nodeIndex++;state.answered=false;renderNode()}else finishCase();
}

function button(text,cls='primary'){
 const b=document.createElement('button');b.className='btn '+cls;b.textContent=text;return b;
}

function renderNode(){
 updateProgress();scrollTop();
 const s=SCENARIOS[state.scenarioIndex], node=s.nodes[state.nodeIndex], box=$('dialogueScroll');
 box.innerHTML='';
 if(node.type==='story') renderStory(node,box);
 else if(node.type==='choice') renderChoice(node,box);
 else if(node.type==='quiz') renderQuiz(node,box);
 else if(node.type==='multi') renderMulti(node,box);
 else if(node.type==='lesson') renderLesson(node,box);
 else {box.textContent='장면을 불러오는 중 문제가 생겼습니다.';setPill('오류');}

}
